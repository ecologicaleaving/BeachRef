### Temporal Filtering Logic Implementation
[Source: Epic 5 specification and existing tournament data structure]

**Tournament Temporal Status Calculation:**
```typescript
// Core temporal filtering utilities to implement
interface TournamentTemporalStatus {
  status: 'active' | 'upcoming' | 'past'
  daysFromNow: number
  displayText: string
  priority: number
}

export function calculateTournamentTemporalStatus(
  tournament: Tournament, 
  currentDate: Date = new Date()
): TournamentTemporalStatus {
  const startDate = new Date(tournament.startDate)
  const endDate = new Date(tournament.endDate)
  const now = currentDate.getTime()
  
  const daysDiff = Math.floor((startDate.getTime() - now) / (1000 * 60 * 60 * 24))
  
  // Active tournament (currently happening)
  if (now >= startDate.getTime() && now <= endDate.getTime()) {
    return {
      status: 'active',
      daysFromNow: 0,
      displayText: 'Active Now',
      priority: 1
    }
  }
  
  // Upcoming tournament
  if (now < startDate.getTime()) {
    return {
      status: 'upcoming',
      daysFromNow: daysDiff,
      displayText: daysDiff === 1 ? 'Starts tomorrow' : `Starts in ${daysDiff} days`,
      priority: 2
    }
  }
  
  // Past tournament
  const daysAgo = Math.abs(daysDiff)
  return {
    status: 'past',
    daysFromNow: daysDiff,
    displayText: daysAgo === 1 ? 'Ended yesterday' : `Ended ${daysAgo} days ago`,
    priority: 3
  }
}

export function filterTournamentsByTimelineRange(
  tournaments: Tournament[],
  currentDate: Date = new Date(),
  range: number = 20
): {
  active: Tournament[]
  upcoming: Tournament[]
  past: Tournament[]
  total: Tournament[]
} {
  const sorted = tournaments
    .map(tournament => ({
      ...tournament,
      temporalStatus: calculateTournamentTemporalStatus(tournament, currentDate)
    }))
    .sort((a, b) => {
      // Sort by priority first, then by date proximity
      if (a.temporalStatus.priority !== b.temporalStatus.priority) {
        return a.temporalStatus.priority - b.temporalStatus.priority
      }
      return Math.abs(a.temporalStatus.daysFromNow) - Math.abs(b.temporalStatus.daysFromNow)
    })
  
  const active = sorted.filter(t => t.temporalStatus.status === 'active')
  const upcoming = sorted.filter(t => t.temporalStatus.status === 'upcoming').slice(0, range)
  const past = sorted.filter(t => t.temporalStatus.status === 'past').slice(0, range)
  
  return { active, upcoming, past, total: [...active, ...upcoming, ...past] }
}
```

interface EnhancedTournamentCardProps {

        <div className="flex items-center justify-between">
