### Tournament Context Analysis Implementation
[Source: Story 4.3 schedule API integration and tournament data structure]

**Smart Default Tab Selection Logic:**
```typescript
// New utility: utils/tournament-context-analysis.ts
interface TournamentContextAnalysis {
  recommendedTab: 'overview' | 'schedule' | 'results'
  reason: 'has_matches' | 'no_matches' | 'tournament_complete' | 'user_preference' | 'loading'
  confidence: 'high' | 'medium' | 'low'
  hasMatches: boolean
  isCompleted: boolean
}

export function analyzeTournamentContext(
  tournament: TournamentDetail,
  userPreference?: string
): TournamentContextAnalysis {
  // Priority 1: User preference override
  if (userPreference && isValidTabId(userPreference)) {
    return {
      recommendedTab: userPreference as TabId,
      reason: 'user_preference',
      confidence: 'high',
      hasMatches: tournament.matches?.length > 0,
      isCompleted: tournament.status === 'completed'
    }
  }

  // Priority 2: Tournament completion status
  if (tournament.status === 'completed') {
    return {
      recommendedTab: 'results',
      reason: 'tournament_complete',
      confidence: 'high',
      hasMatches: tournament.matches?.length > 0,
      isCompleted: true
    }
  }

  // Priority 3: Match availability
  const hasScheduledMatches = tournament.matches && tournament.matches.length > 0
  if (hasScheduledMatches) {
    return {
      recommendedTab: 'schedule',
      reason: 'has_matches',
      confidence: 'high',
      hasMatches: true,
      isCompleted: false
    }
  }

  // Default: Overview tab for tournaments without matches
  return {
    recommendedTab: 'overview',
    reason: 'no_matches',
    confidence: 'medium',
    hasMatches: false,
    isCompleted: false
  }
}
```

### Enhanced TournamentDetailTabs with Context-Aware Selection
[Source: Existing TournamentDetailTabs.tsx and Epic 5.3 requirements]

**Enhanced Component Implementation:**
```typescript
// Enhanced components/tournament/TournamentDetailTabs.tsx
import { analyzeTournamentContext } from '@/utils/tournament-context-analysis'
import { useUserTabPreferences } from '@/hooks/useUserTabPreferences'

export function TournamentDetailTabs({ tournament }: TournamentDetailTabsProps) {
  const router = useRouter()
  const searchParams = useSearchParams()
  const { userPreference, setUserPreference } = useUserTabPreferences()
  const [activeTab, setActiveTab] = useState<TabId>('overview')
  const [contextAnalysis, setContextAnalysis] = useState<TournamentContextAnalysis | null>(null)

  // Smart default tab selection with context analysis
  useEffect(() => {
    if (!tournament) return

    const analysis = analyzeTournamentContext(tournament, userPreference)
    setContextAnalysis(analysis)

    // Check for explicit URL tab parameter (highest priority)
    const urlTab = searchParams?.get('tab')
    if (urlTab && isValidTabId(urlTab)) {
      setActiveTab(urlTab)
      return
    }

    // Use smart default from context analysis
    setActiveTab(analysis.recommendedTab)
  }, [tournament, userPreference, searchParams])

  // Handle tab changes with URL updates
  const handleTabChange = (newTab: TabId) => {
    setActiveTab(newTab)
    
    // Update URL without page reload
    const params = new URLSearchParams(searchParams?.toString())
    params.set('tab', newTab)
    router.replace(`/tournament/${tournament.code}?${params.toString()}`, { scroll: false })
    
    // Track tab selection for analytics
    tabPerformanceMonitor.trackTabSwitch(newTab, contextAnalysis?.reason || 'manual')
  }

  return (
    <div className="space-y-4">
      {/* Context indicator for development/debugging */}
      {process.env.NODE_ENV === 'development' && contextAnalysis && (
        <div className="text-xs text-muted-foreground">
          Smart default: {contextAnalysis.recommendedTab} ({contextAnalysis.reason})
        </div>
      )}
      
      <Tabs value={activeTab} onValueChange={handleTabChange} className="w-full">
        <TabsList className="grid w-full grid-cols-3 h-12">
          {AVAILABLE_TABS.map((tab) => (
            <TabsTrigger
              key={tab.id}
              value={tab.id}
              className="flex items-center gap-2 min-h-[48px] touch-target"
              aria-describedby={`${tab.id}-description`}
            >
              <tab.icon className="h-4 w-4" />
              <span className="hidden sm:inline">{tab.label}</span>
              <span className="sm:hidden">{tab.label.slice(0, 3)}</span>
            </TabsTrigger>
          ))}
        </TabsList>
        
        {/* Tab content with enhanced error boundaries */}
        <TabsContent value="overview" className="mt-4">
          <TabErrorBoundary tabName="Overview">
            <TournamentOverviewTab tournament={tournament} />
          </TabErrorBoundary>
        </TabsContent>
        
        <TabsContent value="schedule" className="mt-4">
          <TabErrorBoundary tabName="Schedule">
            <Suspense fallback={<TournamentScheduleSkeleton />}>
              <TournamentScheduleTab tournament={tournament} />
            </Suspense>
          </TabErrorBoundary>
        </TabsContent>
        
        <TabsContent value="results" className="mt-4">
          <TabErrorBoundary tabName="Results">
            <Suspense fallback={<TournamentResultsSkeleton />}>
              <TournamentResultsTab tournament={tournament} />
            </Suspense>
          </TabErrorBoundary>
        </TabsContent>
      </Tabs>
    </div>
  )
}
```

### User Preference System Implementation
[Source: Epic 5.3 specification and localStorage patterns]

**User Preference Hook:**
```typescript
// New hook: hooks/useUserTabPreferences.ts
import { useState, useEffect } from 'react'

interface UserTabPreferences {
  defaultTab?: 'overview' | 'schedule' | 'results' | 'smart'
  enableSmartDefaults: boolean
  lastUpdated: number
}

export function useUserTabPreferences() {
  const [preferences, setPreferences] = useState<UserTabPreferences>({
    enableSmartDefaults: true,
    lastUpdated: Date.now()
  })

  // Load preferences from localStorage on mount
  useEffect(() => {
    try {
      const stored = localStorage.getItem('beachref-tab-preferences')
      if (stored) {
        const parsed = JSON.parse(stored)
        setPreferences(parsed)
      }
    } catch (error) {
      console.warn('Failed to load tab preferences:', error)
    }
  }, [])

  // Save preferences to localStorage
  const updatePreferences = (updates: Partial<UserTabPreferences>) => {
    const newPreferences = {
      ...preferences,
      ...updates,
      lastUpdated: Date.now()
    }
    
    setPreferences(newPreferences)
    
    try {
      localStorage.setItem('beachref-tab-preferences', JSON.stringify(newPreferences))
    } catch (error) {
      console.warn('Failed to save tab preferences:', error)
    }
  }

  return {
    preferences,
    updatePreferences,
    userPreference: preferences.enableSmartDefaults ? undefined : preferences.defaultTab
  }
}
```
